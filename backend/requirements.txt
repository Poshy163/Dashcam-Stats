# --- inference -------------------------------------------------------------------------
# The OpenVINO build of onnxruntime, not the plain one. The plain build offers only the
# CPU provider, which left every detection and plate model running on the CPU while the
# iGPU idled -- visible in intel_gpu_top as a busy Python process and a quiet render
# engine, and worth roughly thirty-three hours on a 674-recording queue.
#
# ``onnxruntime-openvino`` and the standalone ``openvino`` package load the same native
# OpenVINO libraries into one process. They must be the exact pair from ONNX Runtime's
# compatibility table: ORT 1.24.1 is built for OpenVINO 2025.4.1. Broad ranges previously
# allowed ORT 1.23.0 (built for OpenVINO 2025.3) beside OpenVINO 2025.4.1; the provider was
# listed as available but failed at session creation with an undefined C++ symbol, then
# silently ran every model on CPU.
openvino==2025.4.1
onnxruntime-openvino==1.24.1

# Model weights and their inference code, both MIT. Fetching weights from the projects
# that publish them replaces release assets on this repository that never existed, and
# using their inference removes hand-written letterboxing, NMS and CTC decoding that had
# never run against real weights.
open-image-models>=0.6,<1
fast-plate-ocr>=1.1,<2
