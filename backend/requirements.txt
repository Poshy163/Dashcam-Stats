# Runtime dependencies.
#
# Upper bounds are deliberate: this image is rebuilt automatically, and a silent major
# version bump in SQLAlchemy or Pydantic breaking a user's running deployment is worse
# than being a release behind.

# --- web -------------------------------------------------------------------------------
fastapi>=0.115,<0.117
uvicorn[standard]>=0.32,<0.36
python-multipart>=0.0.12,<0.1

# --- data ------------------------------------------------------------------------------
SQLAlchemy[asyncio]>=2.0.36,<2.1
aiosqlite>=0.20,<0.22
alembic>=1.14,<1.17

pydantic>=2.9,<3
pydantic-settings>=2.6,<3

# --- media / imaging -------------------------------------------------------------------
# headless: the container has no X11, and the GUI build pulls in a large dependency tree
# for functions that are never called here.
opencv-python-headless>=4.10,<5
numpy>=1.26,<3
Pillow>=11.0,<12

# --- inference -------------------------------------------------------------------------
# Execute ONNX models directly with OpenVINO. The separate ONNX Runtime OpenVINO wheel is
# intentionally not installed: its newest Linux build still bundles OpenVINO 2025.4.1 and
# cannot safely coexist with the newer runtime because both packages ship ABI-distinct
# libraries under the same SONAMEs. Plain ONNX Runtime remains for the public types used by
# the upstream preprocessing/postprocessing libraries and as a CPU-only emergency fallback.
openvino==2026.3.0
onnxruntime==1.27.0

# Model weights and their inference code, both MIT. Fetching weights from the projects
# that publish them replaces release assets on this repository that never existed, and
# using their inference removes hand-written letterboxing, NMS and CTC decoding that had
# never run against real weights.
open-image-models>=0.6,<1
fast-plate-ocr>=1.1,<2

# --- support ---------------------------------------------------------------------------
structlog>=24.4,<26
httpx>=0.27,<0.29
# python:*-slim ships no zoneinfo database, and every timestamp in this app is
# timezone-aware. Without this, ZoneInfo("Australia/Adelaide") raises at runtime.
tzdata>=2024.2
